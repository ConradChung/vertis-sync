# Lume Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Read `docs/lume/INVENTORY.md` and `docs/lume/2026-10-14-lume-design.md` first. Before writing any agent code, invoke the `claude-api` skill.

**Goal:** Ship Lume, a second Next.js app at `lume/`, that shows every pipeline gate in one place and taps the same atomic claims Telegram taps, with an events feed, a rotation-rule registry, and an orb agent whose high-risk actions always stop for confirmation.

**Architecture:** Lume reads existing tables through `lume_*` views, writes through service-role route handlers, and reaches the pipelines through one additive entrypoint on `telegram-inbox` (`X-Lume-Secret` → `handleCallback`). New state lives in `events`, `pending_actions`, `agent_runs`, `rotation_rules`, `lume_campaigns`, `lume_external_schedules`. Triggers write `events` and fire pg_net hooks so no existing worker function changes.

**Tech Stack:** Next.js 15.5 App Router, React 19, TypeScript, Tailwind 4, shadcn/ui (new-york, dark), glasscn, Magic UI (NumberTicker, AnimatedList, BorderBeam), `motion`, Recharts via shadcn charts, `@xyflow/react`, `three` + `@react-three/fiber` + `@react-three/drei`, lucide, `@supabase/ssr` + `@supabase/supabase-js`, `@anthropic-ai/sdk` (Tool Runner, `betaZodTool`), `zod`, `cron-parser`, Vitest, pgTAP (`supabase test db`), Deno test.

**Conventions carried over:** migrations are applied with the Supabase MCP `apply_migration` first and the file is then named after the ledger version it was assigned (`select version, name from supabase_migrations.schema_migrations order by version desc limit 1`), e.g. `20261014102012_lume_core.sql`; the `2026101400000N` names below are placeholders; edge functions are self-contained Deno with raw PostgREST; commit per task with the task name; **do not push**; secrets only via `supabase secrets set` / Vercel env, names listed in `.env.example`.

```sql
-- supabase/migrations/20261014000001_lume_core.sql
-- Lume: the command center's own tables. Everything else Lume shows is a
-- view over tables that already exist (see docs/lume/2026-10-14-lume-design.md).

-- Append-only feed behind the Overview. Written by triggers (next migration)
-- and by Lume's own routes (webhooks, agent, decisions). Never updated.
create table if not exists events (
  id          bigserial primary key,
  type        text not null,            -- 'validation.status', 'ingest.approved', 'reply', 'bounce', 'agent.tool', ...
  source      text,                     -- table or integration: 'validation_jobs', 'instantly', 'lume', 'agent'
  source_id   text,
  campaign_id text,                     -- Instantly campaign id when known
  lead_email  text,
  title       text not null,
  payload     jsonb not null default '{}'::jsonb,
  created_at  timestamptz not null default now()
);
create index if not exists events_created_idx on events (created_at desc);
create index if not exists events_type_idx on events (type, created_at desc);
-- Webhook replays (Instantly retries) must not double-count. (Applied as a
-- real column + partial unique index in 20261014103940_lume_core_fixes.sql.)
-- webhook_id text; create unique index events_webhook_id_idx on events (webhook_id) where webhook_id is not null;

-- High-risk actions wait here. claim_status is the atomic-claim slot, same
-- pattern as validation_jobs.finder_status: the decider PATCHes
-- ...&claim_status=eq.awaiting -> claimed and acts only if a row came back.
create table if not exists pending_actions (
  id                  uuid primary key default gen_random_uuid(),
  kind                text not null check (kind in (
                        'push_to_instantly','launch_campaign','assign_leads','activate_rule',
                        'send_reply','pause_campaign','resume_campaign','create_campaign',
                        'decide_gate','delete_rule')),
  payload             jsonb not null default '{}'::jsonb,
  risk                text not null default 'high' check (risk in ('high')),
  status              text not null default 'awaiting'
                      check (status in ('awaiting','approved','rejected','executed','failed','expired')),
  claim_status        text not null default 'awaiting' check (claim_status in ('awaiting','claimed')),
  requested_by        text not null default 'user' check (requested_by in ('user','agent','rule')),
  agent_run_id        uuid,
  rule_id             uuid,
  telegram_message_id bigint,
  decided_via         text check (decided_via in ('lume','telegram')),
  decided_at          timestamptz,
  result              jsonb,
  error               text,
  created_at          timestamptz not null default now(),
  updated_at          timestamptz not null default now()
);
create index if not exists pending_actions_open_idx on pending_actions (status) where status = 'awaiting';
drop trigger if exists set_pending_actions_updated_at on pending_actions;
create trigger set_pending_actions_updated_at before update on pending_actions
  for each row execute function update_updated_at_column();   -- from 20260322000001

create table if not exists agent_runs (
  id                 uuid primary key default gen_random_uuid(),
  prompt             text not null,
  status             text not null default 'running'
                     check (status in ('running','needs_approval','done','failed')),
  plan               jsonb,
  tool_calls         jsonb not null default '[]'::jsonb,
  approval_action_id uuid references pending_actions(id),
  outcome            text,
  model              text,
  usage              jsonb,
  created_at         timestamptz not null default now(),
  finished_at        timestamptz
);
alter table pending_actions
  add constraint pending_actions_agent_run_fk foreign key (agent_run_id) references agent_runs(id);

-- Cron-driven assignment rules. The tick never pushes: it creates a
-- pending_actions row (kind assign_leads) and pings Telegram.
create table if not exists rotation_rules (
  id           uuid primary key default gen_random_uuid(),
  name         text not null,
  campaign_id  text not null,                   -- lume_campaigns.external_id
  filter       jsonb not null default '{}'::jsonb,  -- {source, icp_status, city, since}
  daily_cap    int  not null default 40 check (daily_cap > 0),
  order_by     text not null default 'newest' check (order_by in ('newest','oldest')),
  cron         text not null,
  active       boolean not null default false,
  next_run_at  timestamptz,
  last_run_at  timestamptz,
  last_result  jsonb,
  created_at   timestamptz not null default now()
);
alter table pending_actions
  add constraint pending_actions_rule_fk foreign key (rule_id) references rotation_rules(id) on delete set null;

-- Unified campaign registry. `campaigns` already exists for the CRM, hence the prefix.
create table if not exists lume_campaigns (
  id                    uuid primary key default gen_random_uuid(),
  family                text not null check (family in ('kaspen','recruiter','muse','other')),
  name                  text not null,
  external_id           text,                   -- Instantly campaign id; null for gmail/muse
  sender_type           text not null check (sender_type in ('instantly','gmail','muse')),
  status                text not null default 'unknown',   -- draft|active|paused|completed|manual|unknown
  capacity_per_day      int,
  opportunity_value_usd numeric,
  instantly             jsonb,                  -- raw campaign object from the last sync
  synced_at             timestamptz,
  created_at            timestamptz not null default now(),
  unique (sender_type, external_id)
);

-- Crons that live outside pg_cron (Claude scheduled tasks, Vercel crons).
create table if not exists lume_external_schedules (
  id           uuid primary key default gen_random_uuid(),
  name         text not null unique,
  cron         text not null,
  target       text not null,
  runner       text not null check (runner in ('claude','vercel','other')),
  last_status  text,
  last_run_at  timestamptz,
  notes        text,
  created_at   timestamptz not null default now()
);
insert into lume_external_schedules (name, cron, target, runner, notes) values
  ('job-scrape',    '0 9 */2 * *', 'jobsearch-notify',  'claude', 'Every other day; inserts job_batches then POSTs jobsearch-notify'),

-- RLS: on everywhere; the one authenticated Lume user may read the three
-- tables the browser subscribes to. Every write goes through the service role.
alter table events enable row level security;
alter table pending_actions enable row level security;
alter table agent_runs enable row level security;
alter table rotation_rules enable row level security;
alter table lume_campaigns enable row level security;
alter table lume_external_schedules enable row level security;
create policy events_auth_read on events for select to authenticated using (true);
create policy pending_actions_auth_read on pending_actions for select to authenticated using (true);
create policy agent_runs_auth_read on agent_runs for select to authenticated using (true);

do $$ begin
  if not exists (select 1 from pg_publication_tables where pubname='supabase_realtime' and tablename='events') then
    alter publication supabase_realtime add table public.events;
  end if;
  if not exists (select 1 from pg_publication_tables where pubname='supabase_realtime' and tablename='pending_actions') then
    alter publication supabase_realtime add table public.pending_actions;
  end if;
  if not exists (select 1 from pg_publication_tables where pubname='supabase_realtime' and tablename='agent_runs') then
    alter publication supabase_realtime add table public.agent_runs;

end $$;
```

```sql
-- supabase/migrations/20261014000002_lume_views.sql
-- Read models for Lume. Views, never copies: the pipelines keep writing their
-- own tables and Lume sees the change on the next query / Realtime event.

create or replace view lume_gates as
  select 'ingest_approval'::text as kind, 'telegram_ingests'::text as target_table, i.id::text as target_id,
         i.filename as title,
         format('%s rows · from %s · email column %s', i.row_count, coalesce(i.sender_name,'?'), coalesce(i.detected_email_column,'not detected')) as subtitle,
         case when i.detected_email_column is null then array['deny'] else array['approve','approve_enrich','deny'] end as actions,
         jsonb_build_object('row_count', i.row_count, 'sender_platform', i.sender_platform) as counts,
         i.telegram_message_id, i.created_at
  from telegram_ingests i where i.status = 'awaiting_approval'
union all
  select 'finder_decision', 'validation_jobs', j.id::text, j.filename,
         format('%s rows · %s without an email', j.total_rows, j.rows_missing_email),
         array['finder_go','finder_skip'],
         jsonb_build_object('total_rows', j.total_rows, 'rows_missing_email', j.rows_missing_email, 'enrich', j.enrich),
         null::bigint, j.created_at
  from validation_jobs j where j.finder_status = 'awaiting_decision'
union all
  select 'campaign_choice', 'validation_jobs', j.id::text, j.filename,
         format('%s valid leads · pick an Instantly campaign', j.valid_count),
         array['pick_campaign','push_cancel'],
         jsonb_build_object('valid_count', j.valid_count, 'candidates', j.instantly_campaign_candidates),
         j.instantly_prompt_message_id, j.updated_at
  from validation_jobs j where j.instantly_status in ('awaiting_campaign','choosing_campaign')
union all
  select 'push_review', 'validation_jobs', j.id::text, j.filename,
         format('%s valid → %s', j.valid_count, j.instantly_campaign_name),
         array['bar_loc','bar_co','push_strict','push_std','push_go','push_cancel'],
         jsonb_build_object('valid_count', j.valid_count, 'campaign_id', j.instantly_campaign_id, 'campaign_name', j.instantly_campaign_name,
                            'location_barrier', j.location_barrier, 'company_barrier', j.company_barrier, 'company_strict', j.company_strict,
                            'nolocation_campaign_id', j.instantly_campaign_id_nolocation),
         null::bigint, j.updated_at
  from validation_jobs j where j.instantly_status in ('configuring','reviewing_company')
union all
  select 'stall', 'validation_jobs', j.id::text, j.filename,
         format('stalled at %s/%s', j.processed_rows, j.total_rows),
         array['wd_resume'],
         jsonb_build_object('processed_rows', j.processed_rows, 'total_rows', j.total_rows, 'status', j.status),
         null::bigint, j.watchdog_notified_at
  from validation_jobs j where j.watchdog_status = 'awaiting'
union all
  select 'jobs_review', 'job_batches', b.id::text, format('Job scrape %s', to_char(b.batch_date,'DD Mon')),
         format('%s companies added', b.companies_added),
         array['jobs_pull','jobs_skip'],
         jsonb_build_object('companies_added', b.companies_added, 'lookback_days', b.lookback_days),
         b.telegram_message_id, b.created_at
  from job_batches b where b.status = 'awaiting_review' and b.review_status = 'awaiting'
union all
  select 'jobs_fill', 'job_batches', b.id::text, format('Job batch %s', to_char(b.batch_date,'DD Mon')),
         format('%s companies resolved · below the 500-contact minimum', b.companies_pulled),
         array['jobs_order_now','jobs_abort'],
         jsonb_build_object('companies_pulled', b.companies_pulled, 'fill_rounds', b.fill_rounds),
         b.telegram_message_id, b.updated_at
  from job_batches b where b.status = 'waiting_fill'
union all
  select 'jobs_write', 'job_batches', b.id::text, format('Contacts ready · %s', to_char(b.batch_date,'DD Mon')),
         format('%s kept of %s', b.contacts_kept, b.contacts_total),
         array['jobs_write','jobs_discard'],
         jsonb_build_object('contacts_kept', b.contacts_kept, 'contacts_total', b.contacts_total, 'order_id', b.leadsfriday_order_id),
         null::bigint, b.updated_at
  from job_batches b where b.status = 'ranked' and b.write_status = 'awaiting'
union all
  select 'jobs_stall', 'job_batches', b.id::text, format('Job batch %s stalled', to_char(b.batch_date,'DD Mon')),
         format('at %s', b.status),
         array['jobs_resume','jobs_abort'],
         jsonb_build_object('status', b.status),
         null::bigint, b.watchdog_notified_at
  from job_batches b where b.watchdog_status = 'awaiting'
union all
  select 'lfo_count', 'apollo_orders', o.id::text, 'Apollo search → Leadsfriday',
         'How many contacts? Minimum 500',
         array['lfo_count','lfo_cancel'],
         jsonb_build_object('apollo_url', o.apollo_url),
         o.telegram_message_id, o.created_at
  from apollo_orders o where o.status = 'awaiting_count' and o.count_status = 'awaiting'
union all
  select 'lume_action', 'pending_actions', a.id::text, a.kind,
         coalesce(a.payload->>'summary', a.kind),
         array['lume_ok','lume_no'],
         a.payload,
         a.telegram_message_id, a.created_at
  from pending_actions a where a.status = 'awaiting';

create or replace view lume_orders as
  select 'ingest'::text as kind, i.id::text as id, i.sender_platform as source, i.filename as summary,
         i.row_count, i.status, null::numeric as cost_usd,
         coalesce(i.sender_name, i.sender_platform) as created_by, i.validation_job_id, i.created_at
  from telegram_ingests i
union all
  select 'apollo_order', o.id::text, 'apollo_link', coalesce(o.apollo_url, format('Leadsfriday #%s', o.leadsfriday_order_id)),
         o.contact_count, o.status, o.credits_deducted, 'owner', (select validation_job_id from telegram_ingests t where t.id = o.ingest_id), o.created_at
  from apollo_orders o
union all
  select 'job_batch', b.id::text, 'leadsfriday', format('Job scrape %s · %s companies', b.batch_date, b.companies_pulled),
         b.contacts_total, b.status, null, 'scheduled task', null, b.created_at
  from job_batches b
union all
  select 'dashboard_job', j.id::text, j.source, j.filename, j.total_rows, j.status, null, 'dashboard', j.id, j.created_at
  from validation_jobs j where j.source in ('csv','apify','hnwi');

-- Validated, not yet assigned anywhere. Two sources: the rotation bank's own
-- queue, and valid rows from finished jobs that were never pushed.
create or replace view lume_lead_bank as
  select 'lead_bank'::text as origin, l.id::text as id, l.email, l.first_name, l.last_name, l.company_name,
         l.location as city, null::text as icp_status, l.campaign_id, l.state, l.created_at
  from lead_bank l where l.state = 'queued'
union all
  select 'validation_rows', r.id::text, r.email,
         coalesce(r.row_data->>'First Name', r.row_data->>'first_name'),
         coalesce(r.row_data->>'Last Name',  r.row_data->>'last_name'),
         coalesce(r.row_data->>'Company Name', r.row_data->>'company_name', r.row_data->>'Company'),
         r.row_data->>'operating_city', r.row_data->>'icp_status', null, 'unassigned', r.processed_at
  from validation_rows r
  join validation_jobs j on j.id = r.job_id
  where r.status = 'valid' and r.instantly_status = 'skipped' and j.status = 'completed'
    and not exists (select 1 from lead_bank l where lower(l.email) = lower(r.email));

-- cron.job is not readable by the API roles; expose it the way rotation_dashboard() does.
create or replace function lume_scheduled_jobs() returns json
language sql security definer set search_path = public, pg_catalog stable as $$
  with pg as (
    select j.jobname as name, j.schedule as cron, 'pg_cron'::text as runner, j.active,
           left(j.command, 160) as target,
           (select r.status from cron.job_run_details r where r.jobid = j.jobid order by r.start_time desc limit 1) as last_status,
           (select r.start_time from cron.job_run_details r where r.jobid = j.jobid order by r.start_time desc limit 1) as last_run_at
    from cron.job j),
  ext as (
    select name, cron, runner, true as active, target, last_status, last_run_at from lume_external_schedules),
  rules as (
    select 'rule: ' || name, cron, 'rotation_rule', active, campaign_id, last_result->>'status', last_run_at from rotation_rules)
  select coalesce(json_agg(row_to_json(x)), '[]'::json)
  from (select * from pg union all select * from ext union all select * from rules) x;
$$;
revoke all on function lume_scheduled_jobs() from public;
grant execute on function lume_scheduled_jobs() to service_role;

-- One round trip for the KPI tiles that come from the DB.
create or replace function lume_overview() returns json
language sql security definer set search_path = public, pg_catalog stable as $$
  select json_build_object(
    'now', now(),
    'pending_gates', (select count(*) from lume_gates),
    'replies_today', (select count(*) from events where type = 'reply' and created_at > date_trunc('day', now())),
    'bounces_24h', (select count(*) from events where type = 'bounce' and created_at > now() - interval '24 hours'),
    'active_jobs', (select count(*) from validation_jobs where status in ('pending','processing')),
    'lead_bank_queued', (select count(*) from lead_bank where state = 'queued'),
    'recent_events', (select coalesce(json_agg(e order by e.created_at desc), '[]'::json)
                      from (select * from events order by created_at desc limit 50) e)
  );
$$;
revoke all on function lume_overview() from public;
grant execute on function lume_overview() to service_role;
```

```sql
-- Events without touching worker code: one row per watched-column change.
create or replace function lume_event_on_change() returns trigger
language plpgsql security definer as $$
declare
  cols text[] := tg_argv;      -- watched columns passed at trigger creation
  c text; oldv text; newv text; title text;
  n jsonb := to_jsonb(new); o jsonb := case when tg_op = 'UPDATE' then to_jsonb(old) else '{}'::jsonb end;
begin
  if tg_op = 'INSERT' then
    insert into events (type, source, source_id, campaign_id, lead_email, title, payload)
    values (tg_table_name || '.inserted', tg_table_name, (n->>'id'), n->>'instantly_campaign_id', n->>'email',
            coalesce(n->>'filename', n->>'batch_date', n->>'apollo_url', n->>'kind', tg_table_name), n);
    return new;
  end if;
  foreach c in array cols loop
    oldv := o->>c; newv := n->>c;
    if oldv is distinct from newv then
      title := format('%s %s: %s → %s', tg_table_name, c, coalesce(oldv,'∅'), coalesce(newv,'∅'));
      insert into events (type, source, source_id, campaign_id, lead_email, title, payload)

              title, jsonb_build_object('column', c, 'from', oldv, 'to', newv, 'row', n));
    end if;
  end loop;
  return new;
end $$;
```

```ts
export const LUME_MIN_COUNT = 500
export interface LumeAction { action: string; targetId: string; extra?: string }
export function parseLumeAction(data: string): LumeAction | null {
  const [action, targetId, extra] = data.split(':')
  if (!action || !targetId || !/^[0-9a-f-]{36}$/i.test(targetId)) return null
  if (action === 'lfo_count' && (!extra || Number(extra) < LUME_MIN_COUNT)) return null
  return { action, targetId, extra }
}
```
